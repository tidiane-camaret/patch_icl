% ---------------------------------------------------------------------------
% Query prior across 2 cascade levels, WITH the model's own input/output flow
% (arch.md §5) -- each level is a compact version of arch_patchset.tex's
% pipeline (image -> Encoder -> feature -> tokens -> attention -> Decoder ->
% predicted mask), stacked as lanes: target-image, target-PRIOR, context
% -image, context-mask. Level 1's prior lane is an empty/dashed placeholder
% (support-mean, no real input yet); level 2's is a real cube, produced by
% CROPPING level 1's own prediction around its centroid. That side-by-side
% contrast IS the figure's point -- the target "catches up" to having a
% second (mask-like) lane, exactly like context always had.
% ---------------------------------------------------------------------------

% Bigger arrows than the shared default
\renewcommand{\archarrow}[3]{%
  \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,#3] #1 -- #2;
}

\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

  % -------------------------------------------------------------------------
  % 2 Cascade Levels (Compact & Simplified)
  % -------------------------------------------------------------------------
  \foreach \L/\xs/\rname in {1/0/{Level 1 -- coarse ($r_1=6$mm)}, 
                             2/11.5/{Level 2 -- fine ($r_2=3$mm)}} {
    \begin{scope}[xshift=\xs cm]
      
      \node[anchor=west,font=\bfseries\Large,black!80] at (0,7.3) {\rname};

      % ---- lane 0: target image (always real) ----
      \node[anchor=east,font=\normalsize] at (-0.2,6.15) {$I^T$};
      \archvolcube{0}{5.5}{1.15}{1.3}{archTargetGreen}{ct}{}
      \archarrow{(1.25,6.15)}{(1.6,6.15)}{archArrowPale}
      \archfunnel{1.7}{5.5}{1.3}{0.75}{0.5}{1}{archEncGray}{Enc}
      % Direct compact route to Attention
      \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,archArrowPale] (2.45,6.15) -| (3.1,5.0) -- (3.5,5.0);

      % ---- lane 1: target PRIOR ----
      \ifthenelse{\equal{\L}{1}}{
        % Level 1: support-mean placeholder
        \fill[black!3] (0,3.8) rectangle (1.15, 5.1);
        \draw[black!25, dashed, line width=1.2pt] (0,3.8) rectangle (1.15, 5.1);
        \node[align=center, font=\scriptsize, black!40] at (0.575, 4.45) {support\\mean};
        % Ghosted routing arrow directly into Attention
        \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,black!20] (1.25,4.45) -| (2.8,4.1) -- (3.5,4.1);
      }{
        % Level 2: REAL prior cube (receives crop from Level 1)
        \node[anchor=east,font=\normalsize] at (-0.2,4.45) {$P^T$};
        \archvolcube{0}{3.8}{1.15}{1.3}{archTargetAccent}{predmask}{}
        \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,archArrowPale] (1.25,4.45) -| (2.8,4.1) -- (3.5,4.1);
      }

      % ---- lane 2: context image (always real) ----
      \node[anchor=east,font=\normalsize] at (-0.2,2.75) {$I^C$};
      \archvolcube{0}{2.1}{1.15}{1.3}{archContextPink}{ct}{}
      \archarrow{(1.25,2.75)}{(1.6,2.75)}{archArrowPale}
      \archfunnel{1.7}{2.1}{1.3}{0.75}{0.5}{1}{archEncGray}{Enc}
      % Direct route to Attention
      \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,archArrowPale] (2.45,2.75) -- (3.5,2.75);

      % ---- lane 3: context mask (always real) ----
      \node[anchor=east,font=\normalsize] at (-0.2,0.9) {$M^C$};
      \archvolcube{0}{0.4}{1.15}{1.0}{archContextAccent}{predmask}{}
      \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,archArrowPale] (1.25,0.9) -| (3.1,1.4) -- (3.5,1.4);


      % -------------------------------------------------------------------------
      % Compact Single-Block Attention
      % -------------------------------------------------------------------------
      % Background and Border
      \fill[archAttnBlue!10, rounded corners=4pt] (3.5, 1.1) rectangle (5.7, 5.9);
      \draw[archAttnBlue!70!black, thin, dashed, rounded corners=4pt] (3.5, 1.1) rectangle (5.7, 5.9);
      
      % Label
      \node[align=center, font=\bfseries\normalsize, archAttnBlue!70!black] at (4.6, 6.25) {attention};

      % Internal 6x2 Token Grid
      \archtokengrid{3.8}{3.5}{3}{1}{0.7}{archTargetGreen}         % Tgt Img
      \ifthenelse{\equal{\L}{1}}{
        \archtokengrid{4.7}{3.5}{3}{1}{0.7}{black!15}              % Tgt Lbl (ghosted for L1)
      }{
        \archtokengrid{4.7}{3.5}{3}{1}{0.7}{archTargetAccent}      % Tgt Lbl (active for L2)
      }
      \archtokengrid{3.8}{1.4}{3}{1}{0.7}{archContextPink}         % Ctx Img
      \archtokengrid{4.7}{1.4}{3}{1}{0.7}{archContextAccent}       % Ctx Lbl


      % -------------------------------------------------------------------------
      % Output & Decoder Routing
      % -------------------------------------------------------------------------
      \archarrow{(5.7, 4.55)}{(6.6, 4.55)}{archArrowPale}
      
      \archfunnel{6.7}{3.9}{1.3}{0.75}{0.5}{-1}{archEncGray}{Dec}
      \archarrow{(7.45, 4.55)}{(7.95, 4.55)}{archArrowPale}
      \archvolcube{8.05}{3.9}{1.15}{1.3}{archTargetGreen}{predmask}{$\hat L^t_{\L}$}

      % Level 1 specific Crop Bounding Box
      \ifthenelse{\equal{\L}{1}}{
        % Dashed box directly on the centroid of Level 1's output mask
        \draw[archCtxYellow, line width=1.5pt, dashed] (8.325, 4.25) rectangle ++(0.6, 0.6);
      }{}
    \end{scope}
  }

  % -------------------------------------------------------------------------
  % The Inter-Level Crop Arrow
  % -------------------------------------------------------------------------
  \draw[-{Latex[length=3.8mm,width=3.4mm]},line width=2.5pt,archCtxYellow!90!black]
    (9.25, 4.65) to[bend left=18] (11.4, 4.45);
  \node[align=center,font=\bfseries\Large,archCtxYellow!90!black] at (10.35, 5.3) {crop};

\end{tikzpicture}
\end{document}
