% ---------------------------------------------------------------------------
% Query prior across 2 cascade levels, WITH the model's own input/output flow
% (arch.md §5) -- each level is a compact version of arch_patchset.tex's
% pipeline (image -> Encoder -> feature -> tokens -> attention -> Decoder ->
% predicted mask), stacked as lanes: target-image, target-PRIOR, context
% -image, context-mask. Level 1's prior lane is an empty/dashed placeholder
% (support-mean, no real input yet); level 2's is a real cube, produced by
% CROPPING level 1's own prediction around its centroid. That side-by-side
% contrast IS the figure's point -- the target "catches up" to having a
% second (mask-like) lane, exactly like context always had.
% ---------------------------------------------------------------------------

% Bigger arrows than the shared default
\renewcommand{\archarrow}[3]{%
  \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,#3] #1 -- #2;
}

\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

  % -------------------------------------------------------------------------
  % 2 Cascade Levels (Level 1: support-mean prior -> Level 2: cropped prior)
  % -------------------------------------------------------------------------
  \foreach \L/\xs/\rname in {1/0/{Level 1 -- coarse ($r_1=6$mm)}, 
                             2/21.0/{Level 2 -- fine ($r_2=3$mm)}} {
    \begin{scope}[xshift=\xs cm]
      
      \node[anchor=west,font=\bfseries\Large,black!80] at (0,7.3) {\rname};

      % ---- lane 0: target image (always real) ----
      \node[anchor=east,font=\normalsize] at (-0.2,6.15) {$I^T$};

      \archarrow{(1.25,6.15)}{(1.6,6.15)}{archArrowPale}
      \archfunnel{1.7}{5.5}{1.3}{0.75}{0.5}{1}{archEncGray}{Enc}
      \archarrow{(3.0,6.15)}{(3.4,6.15)}{archArrowPale}
      \archfeatcube{3.5}{5.6}{0.7}{1.1}{archTargetGreen}{$F^T$}
      % Route down to Target Img tokens

      % ---- lane 1: target PRIOR ----
      \ifthenelse{\equal{\L}{1}}{
        % Level 1: support-mean placeholder
        \fill[black!3] (0,3.8) rectangle (1.15, 5.1);

        \node[align=center, font=\scriptsize, black!40] at (0.575, 4.45) {support\\mean};
        % Ghosted routing arrow
        \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,black!15] (1.25,4.45) -| (5.4,4.55) -- (5.6,4.55);
      }{
        % Level 2: REAL prior cube (receives crop from Level 1)
        \node[anchor=east,font=\normalsize] at (-0.2,4.45) {$P^T$};
        \archvolcube{0}{3.8}{1.15}{1.3}{archTargetAccent}{predmask}{}
        \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,archArrowPale] (1.25,4.45) -| (5.4,4.55) -- (5.6,4.55);
      }

      % ---- lane 2: context image (always real) ----
      \node[anchor=east,font=\normalsize] at (-0.2,2.75) {$I^C$};

      \archarrow{(1.25,2.75)}{(1.6,2.75)}{archArrowPale}
      \archfunnel{1.7}{2.1}{1.3}{0.75}{0.5}{1}{archEncGray}{Enc}
      \archarrow{(3.0,2.75)}{(3.4,2.75)}{archArrowPale}
      \archfeatcube{3.5}{2.2}{0.7}{1.1}{archContextPink}{$F^C$}

      \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,archArrowPale] (4.3,2.75) -| (4.6,2.45) -- (4.9,2.45);

      % ---- lane 3: context mask (always real) ----
      \node[anchor=east,font=\normalsize] at (-0.2,0.9) {$M^C$};
      \archvolcube{0}{0.4}{1.15}{1.0}{archContextAccent}{predmask}{}

      \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,archArrowPale] (1.25,0.9) -| (5.4,2.45) -- (5.6,2.45);


      % -------------------------------------------------------------------------

      % -------------------------------------------------------------------------

      \archtokengrid{8.8}{2.15}{3}{1}{0.45}{archContextAccent!40!white}
      \foreach \y in {2.375, 2.825, 3.275, 3.725, 4.175, 4.625} {
        \draw[{Latex[length=1.2mm,width=1.2mm]}-{Latex[length=1.2mm,width=1.2mm]}, line width=0.8pt, black!50] (7.95, \y) -- (8.7, \y);
      }

      % Sub-box 2: Patch axis
      \draw[black!20, fill=white, rounded corners=4pt, line width=0.8pt] (9.75, 1.25) rectangle ++(2.6, 4.5);

      \archtokengrid{11.4}{3.5}{3}{1}{0.45}{black!15}
