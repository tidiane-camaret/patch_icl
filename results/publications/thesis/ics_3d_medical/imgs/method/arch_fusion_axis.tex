% ---------------------------------------------------------------------------
% Figure B of the former arch_biaxial.tex: the fusion-axis detail within the
% shared-encoder family arch_fusion_family.tex introduces generically. Two
% panels, stacked, both members of that same family (only the fusion axis
% differs between them):
%
% Single axis attention (top): 1x Fuse (image/label tokens merged once per
% cell by one shared operator -- concat / sum / MLP / pixel-shuffle, drawn
% as a single oplus) followed by Lx Position axis -- fusion happens once,
% position-axis attention repeats for L layers, so only the Position-axis
% box carries the "xL" mark. Only one axis (position) is ever attention.
% Dual attention (bottom): the WHOLE pair (modality axis -> position axis)
% repeats as one layer, Lx -- so both boxes sit inside one outer dashed
% wrapper that carries the single "xL" mark for the pair. Both axes
% (modality and position) are attention.
%
% Style follows TabPFN's Fig. 1 (Hollmann et al., Nature 2024): small token
% grids connected by simple curved double-arrows rather than full
% cell/token-block renderings. The "special test row" device (edges
% touching it get a distinct color) maps onto our target row, which really
% is architecturally special in both methods: it cross-attends/fuses
% read-only into context, and its label-stream input is a predicted prior
% rather than real ground truth like context's, which is why target's
% label tokens are gray rather than a role color throughout.
% ---------------------------------------------------------------------------

% \tpsq{x}{y}{fill}{ring} -- one token, a filled square with a role ring.
\newcommand{\tpsq}[4]{
  \fill[#3] (#1-\tsz/2,#2-\tsz/2) rectangle ++(\tsz,\tsz);
  \draw[#4,line width=1.2pt] (#1-\tsz/2,#2-\tsz/2) rectangle ++(\tsz,\tsz);
}
% \tpcluster{x}{y}{fill}{ring} -- 3 stacked tokens standing in for N.
\newcommand{\tpcluster}[4]{
  \foreach \doff in {-\toff,0,\toff} {
    \tpsq{#1}{#2+\doff}{#3}{#4}
  }
}
% \tpsqsplit{x}{y}{fillImg}{fillLbl}{ring} -- one token, diagonally split
\newcommand{\tpsqsplit}[5]{
  \begin{scope}
    \clip (#1-\tsz/2,#2-\tsz/2) rectangle ++(\tsz,\tsz);
    \fill[#3] (#1-\tsz/2,#2-\tsz/2) -- (#1+\tsz/2,#2-\tsz/2) -- (#1+\tsz/2,#2+\tsz/2) -- cycle;
    \fill[#4] (#1-\tsz/2,#2-\tsz/2) -- (#1-\tsz/2,#2+\tsz/2) -- (#1+\tsz/2,#2+\tsz/2) -- cycle;
  \end{scope}
  \draw[#5,line width=1.2pt] (#1-\tsz/2,#2-\tsz/2) rectangle ++(\tsz,\tsz);
}
% \tpclustersplit{x}{y}{fillImg}{fillLbl}{ring} -- 3 stacked split tokens.
\newcommand{\tpclustersplit}[5]{
  \foreach \doff in {-\toff,0,\toff} {
    \tpsqsplit{#1}{#2+\doff}{#3}{#4}{#5}
  }
}
% \dblarrow{p1}{p2}{color} -- curved double-headed attention edge.
\newcommand{\dblarrow}[3]{
  \draw[{Latex[length=3.0mm,width=2.5mm]}-{Latex[length=3.0mm,width=2.5mm]},
        line width=1.5pt,#3] #1 to[bend left=12] #2;
}
% \fusearrow{p1}{p2}{color} -- straight single-headed merge edge.
% Line width matches \archflow's compact bucket (2.2pt) -- only the head
% shape (3.0mm/2.5mm) stays distinct, to read as "merge" not "flow".
\newcommand{\fusearrow}[3]{
  \draw[-{Latex[length=3.0mm,width=2.5mm]},line width=2.2pt,#3] #1 -- #2;
}

% \tabpfnattn{x} -- Fully connected, right-side routed attention with cascading transparency
\newcommand{\tabpfnattn}[1]{
  \def\attnstyle{[{Latex[length=1.0mm,width=0.8mm]}-{Latex[length=1.0mm,width=0.8mm]}, line width=1.0pt, archBrightBlue]}
  \def\ax{#1+\tsz/2+0.05}

  \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=30] (\ax, \yTgt);
  \draw\attnstyle (\ax, \yTgt)       to[bend left=30] (\ax, \yTgt-\toff);
  \draw\attnstyle (\ax, \yTgt-\toff) to[bend left=30] (\ax, \yCtx+\toff);
  \draw\attnstyle (\ax, \yCtx+\toff) to[bend left=30] (\ax, \yCtx);
  \draw\attnstyle (\ax, \yCtx)       to[bend left=30] (\ax, \yCtx-\toff);

  \begin{scope}[opacity=0.6]
    \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=45] (\ax, \yTgt-\toff);
    \draw\attnstyle (\ax, \yTgt)       to[bend left=45] (\ax, \yCtx+\toff);
    \draw\attnstyle (\ax, \yTgt-\toff) to[bend left=45] (\ax, \yCtx);
    \draw\attnstyle (\ax, \yCtx+\toff) to[bend left=45] (\ax, \yCtx-\toff);
  \end{scope}

  \begin{scope}[opacity=0.3]
    \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=60] (\ax, \yCtx+\toff);
    \draw\attnstyle (\ax, \yTgt)       to[bend left=60] (\ax, \yCtx);
    \draw\attnstyle (\ax, \yTgt-\toff) to[bend left=60] (\ax, \yCtx-\toff);
  \end{scope}

  \begin{scope}[opacity=0.15]
    \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=75] (\ax, \yCtx);
    \draw\attnstyle (\ax, \yTgt)       to[bend left=75] (\ax, \yCtx-\toff);
  \end{scope}

  \begin{scope}[opacity=0.08]
    \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=90] (\ax, \yCtx-\toff);
  \end{scope}
}

% ---- dimensions, matched to the other arch_* figures ----

% ---- Enc/Dec bookend funnels -- see the comment at each panel's own
% Enc/Dec block below for why these are here and why Dec is target-only.
% Sized to match the Enc/Dec funnels in arch_fusion_family.tex, for a
% consistent look across the figure family.
\def\encw{0.9}
\def\encwn{0.55}

\begin{tikzpicture}[x=1cm,y=1cm]
% Whole picture scaled down 25% -- shrinks the native canvas (this figure
% was needlessly tall/square vs. its arch_* siblings) and, since TikZ node
% text is NOT rescaled by a canvas transform while everything else (boxes,
% lines, gaps) is, it also makes text ~33% bigger relative to the geometry
% once re-inflated to \linewidth on the page. Tune this one number if the
% balance needs adjusting after a compile.
\begin{scope}[scale=0.75]

  \pgfmathsetmacro{\xB}{\xA+\boxw+\panelgap}
  % \boxAcenter is set later, once \boxwFuse (the Fuse box's own, narrower
  % width) is known
  \pgfmathsetmacro{\boxBcenter}{\xB+\boxw/2}

  \pgfmathsetmacro{\outerLeftBase}{\xB-0.50}
  \pgfmathsetmacro{\outerRightBase}{\xB+\boxw+0.50}

  % Stacked Layout: Ours shares X-coords with Baseline
  \pgfmathsetmacro{\xC}{\xA}
  \pgfmathsetmacro{\xD}{\xB}
  \pgfmathsetmacro{\boxCcenter}{\xC+\boxw/2}
  \pgfmathsetmacro{\boxDcenter}{\xD+\boxw/2}

  \pgfmathsetmacro{\outerLeft}{\xC-0.50}
  \pgfmathsetmacro{\outerRight}{\xD+\boxw+0.50}

  % raised along with \boxTop below -- the "Modality axis"/"Position axis"
  % (and "Fuse"/"Position axis") titles sit above the inner boxes at
  % \boxTop+0.20, so this dashed wrapper needs enough headroom past that
  % to fully contain the title text, not just the inner box itself
  \pgfmathsetmacro{\outerTop}{\yTgt+\clusterhalf+\boxpad+1.10}
  \pgfmathsetmacro{\outerBot}{\yCtx-\clusterhalf-\boxpad-0.40}

  \pgfmathsetmacro{\methodTitleY}{\outerTop+0.50}
  % shared top edge for all 4 inner boxes (Fuse/Position axis in the top
  % panel, Modality axis/Position axis in the bottom panel) -- raised
  % slightly above the plain \boxpad margin so each box's
  % $Z^{\text{img}}$/$Z^{\text{lbl}}$ superscript labels have enough
  % headroom (same fix as arch_fusion_family.tex's Fuse box)
  \pgfmathsetmacro{\boxTop}{\yTgt+\clusterhalf+\boxpad+0.15}

  \pgfmathsetmacro{\xlblTop}{\ximgTop+1.4}
  % Fuse box narrowed to stop shortly after the lbl column. Right margin
  % (past \xlblTop) matches the left margin (\xA to \ximgTop) exactly, so
  % the img/lbl column pair sits centered in the box rather than right-hung
  \pgfmathsetmacro{\boxwFuse}{\xlblTop+\ximgTop-2*\xA}
  \pgfmathsetmacro{\boxAcenter}{\xA+\boxwFuse/2}

  % Baseline panel's Enc/Dec placement -- Enc sits directly left of the
  % (un-wrapped) Fuse box; Dec sits past the Position-axis box's OWN xL
  % wrapper (\outerRightBase), with extra clearance so it doesn't crowd
  % that dashed boundary. Computed here (outer level, not panel-local) so
  % the divider line below the panel can also reach past them.
  \pgfmathsetmacro{\xEncLeft}{\xA-0.60-\encw}
  \pgfmathsetmacro{\xDecLeft}{\outerRightBase+0.60}

  \pgfmathsetmacro{\xFuseOp}{(\ximgTop+\xlblTop)/2}
  \pgfmathsetmacro{\yMid}{\rowpitch/2}

  \pgfmathsetmacro{\xFusedPatchTop}{\boxBcenter}
  \pgfmathsetmacro{\ximgBot}{\boxCcenter-\colpitch/2}
  \pgfmathsetmacro{\xlblBot}{\ximgBot+\colpitch}
  \pgfmathsetmacro{\ximgPatchBot}{\boxDcenter-\pairpitch/2}
  \pgfmathsetmacro{\xlblPatchBot}{\ximgPatchBot+\pairpitch}


  % =========================================================================
  % Baseline: 1x Fuse, then Lx Position axis (TOP PANEL)
  % =========================================================================
  \begin{scope}[yshift=0cm]

    \node[anchor=south] at ({\xA+(\xB+\boxw-\xA)/2},{\methodTitleY}) {\archboxlabel Single axis attention};

    \draw[archTargetGreen!60!black,fill=archTargetGreen!10,dashed,rounded corners=8pt,line width=1pt]
      ({\outerLeftBase},{\outerBot}) rectangle ({\outerRightBase},{\outerTop});

    % Enc/Dec bookends -- this figure is a zoomed-in fusion/attention
    % detail (per the header comment, deliberately no input cubes or
    % encoder/decoder, unlike arch_fusion_family.tex's own token-grid
    % style), so these are a light, secondary reminder of where the
    % tokens shown here sit within the larger Enc -> Fuse/Attn -> Dec
    % pipeline that figure draws in full -- hence the slight fill
    % opacity, to stay visually subordinate to the real content. Dec is
    % target-row only: context is read-only and its post-attention state
    % is never decoded, same rule as every other arch_fusion_*.tex panel.
    \begin{scope}[fill opacity=\encdecOpacity]
      \archfunnel{\xEncLeft}{{\yTgt-\ench/2}}{\ench}{\encw}{\encwn}{1}{archEncGray}{Enc}
      \archfunnel{\xEncLeft}{{\yCtx-\ench/2}}{\ench}{\encw}{\encwn}{1}{archEncGray}{Enc}
      \archfunnel{\xDecLeft}{{\yTgt-\dech/2}}{\dech}{\decw}{\decwn}{-1}{archEncGray}{Dec}
    \end{scope}
    \archflow{({\xEncLeft+\encw+0.05},{\yTgt})}{({\xA-0.05},{\yTgt})}{neutral}
    \archflow{({\xEncLeft+\encw+0.05},{\yCtx})}{({\xA-0.05},{\yCtx})}{neutral}
    \archflow{({\outerRightBase+0.05},{\yTgt})}{({\xDecLeft-0.05},{\yTgt})}{neutral}

    \draw[black!30,fill=white,rounded corners=5pt,line width=0.8pt]
      (\xA,\yCtx-\clusterhalf-\boxpad) rectangle ({\xA+\boxwFuse},{\boxTop});
    \draw[black!30,fill=white,rounded corners=5pt,line width=0.8pt]
      ({\xB},\yCtx-\clusterhalf-\boxpad) rectangle ({\xB+\boxw},{\boxTop});

    \node[anchor=south] at ({\boxAcenter},{\boxTop+0.20}) {\archboxlabel Fuse};
    \node[anchor=south] at ({\boxBcenter},{\boxTop+0.20}) {\archboxlabel Position axis};

    % starts just past the Fuse box's own (narrowed) edge -- no fused
    % token column sits between the two blocks any more. Margins widened
    % (0.30 -> 0.55) to shorten the arrow itself now that the gap is tight.
    \archflow{({\xA+\boxwFuse+0.55},{\yMid})}{({\xB-0.55},{\yMid})}{neutral}

    \node[anchor=south,font=\normalsize] at (\ximgTop,\yTgt+\clusterhalf+0.05) {$Z^{\text{img}}$};
    \node[anchor=south,font=\normalsize] at (\xlblTop,\yTgt+\clusterhalf+0.05) {$Z^{\text{lbl}}$};

    \foreach \x in {\ximgTop, \xlblTop} {
      \foreach \y in {\yTgt, \yCtx} {
        \fill[black!15,rounded corners=3pt]
          ({\x-\tsz/2-0.18},{\y-\toff-\tsz/2-0.10}) rectangle
          ({\x+\tsz/2+0.18},{\y+\toff+\tsz/2+0.10});
      }
    }

    \foreach \y in {\yTgt,\yCtx} {
      \foreach \doff in {-\toff,0,\toff} {
        \fill[black!15,rounded corners=3pt]
          ({\xFusedPatchTop-\tsz/2-0.18},{\y+\doff-\tsz/2-0.10}) rectangle
          ({\xFusedPatchTop+\tsz/2+0.18},{\y+\doff+\tsz/2+0.10});
      }
    }

    % target's label column is gray, not a role color -- it is never real
    % ground truth (a zero-mask placeholder here), unlike context's
    \foreach \y/\imgfill/\lblfill in {\yTgt/archTargetGreen/black!8,
                                       \yCtx/archContextPink/archContextAccent} {
      \tpcluster{\ximgTop}{\y}{\imgfill}{black!70}
      \tpcluster{\xlblTop}{\y}{\lblfill}{black!70}
      \tpclustersplit{\xFusedPatchTop}{\y}{\imgfill}{\lblfill}{black!70}
    }

    \node[font=\huge] at ({\xFuseOp},{\yTgt}) {$\oplus$};
    \node[font=\huge] at ({\xFuseOp},{\yCtx}) {$\oplus$};

    \tabpfnattn{\xFusedPatchTop}

  \end{scope}


  % =========================================================================
  % Divider between Baseline and Ours
  % =========================================================================
  % raised from the old -1.2 offset -- that put it only ~0.5 above the
  % bottom panel's "Dual axis attention" title (methodTitleY + yshift),
  % close enough to visually overlap it; -0.7 keeps it clear of both that
  % title below and the top panel's own wrapper/boxes above
  \pgfmathsetmacro{\divY}{\outerBot - 0.7}
  \draw[black!20,dashed,line width=1pt] ({\xEncLeft-0.30},{\divY}) -- ({\xDecLeft+\decw+0.30},{\divY});


  % =========================================================================
  % Ours: Lx (image/label axis -> patch axis) (BOTTOM PANEL)
  % =========================================================================
  \begin{scope}[yshift=-9.5cm]

    % Compact, panel-local geometry: unlike the Baseline panel above, the
    % two axes here are one paired operation per layer, not two sequential
    % stages -- so there is no connecting arrow between them, and both
    % boxes (and the outer wrapper) are narrowed to just what their own
    % content needs, rather than sharing \boxw/\panelgap with Baseline.
    % Position axis matches Modality axis's width, with its pair of
    % columns (and the \tabpfnattn fan hanging off \xlblPatchBot) centered
    % the same way Modality axis's own img/lbl pair is.
    \pgfmathsetmacro{\boxwMod}{3.8}
    \pgfmathsetmacro{\boxwPos}{\boxwMod}
    \pgfmathsetmacro{\gapOurs}{0.8}
    % Now that this panel is narrower than Baseline's, re-center it (as a
    % whole) under Baseline's overall span instead of sharing Baseline's
    % left edge \xA -- otherwise it reads as left-hung, not a sibling panel.
    \pgfmathsetmacro{\topCenter}{(\xA+\outerRightBase)/2}
    \pgfmathsetmacro{\xC}{\topCenter-(\boxwMod+\gapOurs+\boxwPos)/2}
    \pgfmathsetmacro{\outerLeft}{\xC-0.50}
    \pgfmathsetmacro{\xD}{\xC+\boxwMod+\gapOurs}
    \pgfmathsetmacro{\boxCcenter}{\xC+\boxwMod/2}
    \pgfmathsetmacro{\boxDcenter}{\xD+\boxwPos/2}
    \pgfmathsetmacro{\outerRight}{\xD+\boxwPos+0.50}
    \pgfmathsetmacro{\ximgBot}{\boxCcenter-\colpitch/2}
    \pgfmathsetmacro{\xlblBot}{\ximgBot+\colpitch}
    \pgfmathsetmacro{\ximgPatchBot}{\boxDcenter-\pairpitch/2}
    \pgfmathsetmacro{\xlblPatchBot}{\ximgPatchBot+\pairpitch}

    \node[anchor=south] at ({(\outerLeft+\outerRight)/2},{\methodTitleY}) {\archboxlabel Dual axis attention};

    \draw[archTargetGreen!60!black,fill=archTargetGreen!10,dashed,rounded corners=8pt,line width=1pt]
      ({\outerLeft},{\outerBot}) rectangle ({\outerRight},{\outerTop});

    % Enc/Dec bookends -- see the matching note in the Baseline panel
    % above. Reuses that panel's OWN \xEncLeft/\xDecLeft (not this
    % panel's narrower wrapper) so the two rows' Enc/Dec funnels line up
    % in the same x-columns, top to bottom, like every other shared
    % x-coordinate in this figure (\xA, \yTgt/\yCtx, etc).
    \begin{scope}[fill opacity=\encdecOpacity]
      \archfunnel{\xEncLeft}{{\yTgt-\ench/2}}{\ench}{\encw}{\encwn}{1}{archEncGray}{Enc}
      \archfunnel{\xEncLeft}{{\yCtx-\ench/2}}{\ench}{\encw}{\encwn}{1}{archEncGray}{Enc}
      \archfunnel{\xDecLeft}{{\yTgt-\dech/2}}{\dech}{\decw}{\decwn}{-1}{archEncGray}{Dec}
    \end{scope}
    \archflow{({\xEncLeft+\encw+0.05},{\yTgt})}{({\outerLeft-0.05},{\yTgt})}{neutral}
    \archflow{({\xEncLeft+\encw+0.05},{\yCtx})}{({\outerLeft-0.05},{\yCtx})}{neutral}
    \archflow{({\outerRight+0.05},{\yTgt})}{({\xDecLeft-0.05},{\yTgt})}{neutral}

    \draw[black!30,fill=white,rounded corners=5pt,line width=0.8pt]
      (\xC,\yCtx-\clusterhalf-\boxpad) rectangle ({\xC+\boxwMod},{\boxTop});
    \draw[black!30,fill=white,rounded corners=5pt,line width=0.8pt]
      ({\xD},\yCtx-\clusterhalf-\boxpad) rectangle ({\xD+\boxwPos},{\boxTop});

    \node[anchor=south] at ({\boxCcenter},{\boxTop+0.20}) {\archboxlabel Modality axis};
    \node[anchor=south] at ({\boxDcenter},{\boxTop+0.20}) {\archboxlabel Position axis};

    \node[anchor=south,font=\normalsize] at ({\ximgBot},\yTgt+\clusterhalf+0.05) {$Z^{\text{img}}$};
    \node[anchor=south,font=\normalsize] at ({\xlblBot},\yTgt+\clusterhalf+0.05) {$Z^{\text{lbl}}$};
    \node[anchor=south,font=\normalsize] at ({\ximgPatchBot},\yTgt+\clusterhalf+0.05) {$Z^{\text{img}}$};
    \node[anchor=south,font=\normalsize] at ({\xlblPatchBot},\yTgt+\clusterhalf+0.05) {$Z^{\text{lbl}}$};

    \fill[black!15,rounded corners=3pt]
      ({\ximgBot-\tsz/2-0.18},{\yCtx-\toff-\tsz/2-0.10}) rectangle

      ({\xlblBot-\tsz/2-0.18},{\yCtx-\toff-\tsz/2-0.10}) rectangle

                                       \yCtx/archContextPink/archContextAccent} {
      \tpcluster{\ximgBot}{\y}{\imgfill}{black!70}
      \tpcluster{\xlblBot}{\y}{\lblfill}{black!70}
      \tpcluster{\ximgPatchBot}{\y}{\imgfill}{black!70}
      \tpcluster{\xlblPatchBot}{\y}{\lblfill}{black!70}
    }

    \foreach \doff in {-\toff,0,\toff} {
      \dblarrow{({\ximgBot+\tsz/2+0.10},\yCtx+\doff)}{({\xlblBot-\tsz/2-0.10},\yCtx+\doff)}{archBrightBlue}
      \dblarrow{({\ximgBot+\tsz/2+0.10},\yTgt+\doff)}{({\xlblBot-\tsz/2-0.10},\yTgt+\doff)}{archBrightBlue}
    }

    \tabpfnattn{\xlblPatchBot}

  \end{scope}

\end{scope}
