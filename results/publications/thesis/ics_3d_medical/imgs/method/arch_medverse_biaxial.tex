% ---------------------------------------------------------------------------
% Medverse's single-level fusion, drawn in the token-cluster vocabulary of
% arch_biaxial.tex for the Fuse step, then switching to the funnel/bowtie
% vocabulary of the original arch_medverse.tex for the dual U-Net -- a flat
% repeated box (this figure's earlier draft) is the wrong shape here: U-Net
% stages actually change resolution stage to stage, they are not an
% identical block repeated xL like a transformer layer, so there is no xL
% marker on this panel at all -- the funnel's shape already encodes depth.
%
% Simplification, spelled out rather than drawn literally: the real
% attention is asymmetric between the funnel's two halves --
%   - narrowing (encoder) side: CONTEXT queries TARGET's shortcuts
%     (context_unet's t2c blocks; target's own encoder is plain conv, never
%     queried, never querying)
%   - widening (decoder) side: TARGET queries CONTEXT's decoder outputs
%     (target_decoder's c2t blocks; context's own decoder is plain conv,
%     never queried, never querying)
% i.e. attention couples the two funnels' matching halves diagonally, not
% both streams at every stage. Drawn here as one generic waist-to-waist
% coupling for legibility, with the asymmetry called out in the note below
% instead of drawn exactly.
% ---------------------------------------------------------------------------

\definecolor{archBrightBlue}{HTML}{3B82F6}

% \tpsq{x}{y}{fill}{ring} -- one token, a filled square with a role ring.
\newcommand{\tpsq}[4]{
  \fill[#3] (#1-\tsz/2,#2-\tsz/2) rectangle ++(\tsz,\tsz);
  \draw[#4,line width=1.2pt] (#1-\tsz/2,#2-\tsz/2) rectangle ++(\tsz,\tsz);
}
% \tpcluster{x}{y}{fill}{ring} -- 3 stacked tokens standing in for N tokens.
\newcommand{\tpcluster}[4]{
  \foreach \doff in {-\toff,0,\toff} {
    \tpsq{#1}{#2+\doff}{#3}{#4}
  }
}
% \tpsqsplit{x}{y}{fillImg}{fillLbl}{ring} -- one token, diagonally split.
\newcommand{\tpsqsplit}[5]{
  \begin{scope}
    \clip (#1-\tsz/2,#2-\tsz/2) rectangle ++(\tsz,\tsz);
    \fill[#3] (#1-\tsz/2,#2-\tsz/2) -- (#1+\tsz/2,#2-\tsz/2) -- (#1+\tsz/2,#2+\tsz/2) -- cycle;
    \fill[#4] (#1-\tsz/2,#2-\tsz/2) -- (#1-\tsz/2,#2+\tsz/2) -- (#1+\tsz/2,#2+\tsz/2) -- cycle;
  \end{scope}
  \draw[#5,line width=1.2pt] (#1-\tsz/2,#2-\tsz/2) rectangle ++(\tsz,\tsz);
}
% \tpclustersplit{x}{y}{fillImg}{fillLbl}{ring} -- 3 stacked split tokens.
\newcommand{\tpclustersplit}[5]{
  \foreach \doff in {-\toff,0,\toff} {
    \tpsqsplit{#1}{#2+\doff}{#3}{#4}{#5}
  }
}
% \fusearrow{p1}{p2}{color} -- straight single-headed merge/passthrough edge.
\newcommand{\fusearrow}[3]{
  \draw[-{Latex[length=3.0mm,width=2.5mm]},line width=1.5pt,#3] #1 -- #2;
}
% \waistarrow{p1}{p2}{color} -- curved double-headed coupling edge between
% the two funnels' waists.
\newcommand{\waistarrow}[3]{
  \draw[{Latex[length=3.0mm,width=2.5mm]}-{Latex[length=3.0mm,width=2.5mm]},
        line width=1.5pt,#3] #1 to[bend left=12] #2;
}

% ---- dimensions, matched to arch_biaxial.tex so the figures sit at the
% same visual scale if placed side by side later ----
\def\tsz{0.50}
\def\toff{0.80}
\def\rowpitch{2.40}
\def\panelgap{2.6}
\def\clusterhalf{1.10}

\def\xlab{-0.8}
\def\yTgt{\rowpitch}
\def\yCtx{0}
\def\yMid{\rowpitch/2}

\def\boxw{6.0}
\def\xFuse{-0.30}
\def\ximgTop{1.0}
\def\boxpad{0.60}

% ---- bowtie geometry ----
\def\bw{4.4}
\def\bh{2.0}
\def\bpinch{0.7}

\begin{tikzpicture}[x=1cm,y=1cm]

  \pgfmathsetmacro{\xEnc}{\xFuse+\boxw+\panelgap}
  \pgfmathsetmacro{\xOut}{\xEnc+\boxw+1.6}
  \pgfmathsetmacro{\boxFuseCenter}{\xFuse+\boxw/2}
  \pgfmathsetmacro{\boxEncCenter}{\xEnc+\boxw/2}

  \pgfmathsetmacro{\outerTop}{\yTgt+\clusterhalf+\boxpad+0.85}
  \pgfmathsetmacro{\outerBot}{\yCtx-\clusterhalf-\boxpad-0.40}
  \pgfmathsetmacro{\methodTitleY}{\outerTop+0.50}

  \pgfmathsetmacro{\xlblTop}{\ximgTop+1.4}
  \pgfmathsetmacro{\xFusedTop}{\xlblTop+2.0}
  \pgfmathsetmacro{\xFuseOp}{(\ximgTop+\xlblTop)/2}
  \pgfmathsetmacro{\xFuseArrowStart}{\xFuseOp+0.35}

  % bowtie anchor: bottom-left corner, one per row, centered on that row
  \pgfmathsetmacro{\bcx}{\xEnc+(\boxw-\bw)/2}
  \pgfmathsetmacro{\bcyTgt}{\yTgt-\bh/2}
  \pgfmathsetmacro{\bcyCtx}{\yCtx-\bh/2}
  \pgfmathsetmacro{\waistX}{\bcx+\bw/2}

  % ---- row labels ----
  \node[anchor=east,font=\normalsize] at (\xlab,\yTgt) {target ($N$)};
  \node[anchor=east,font=\normalsize] at (\xlab,\yCtx) {context ($N$)};

  % ---- title ----
  \node[anchor=south] at ({\xFuse+(\xOut-\xFuse)/2},{\methodTitleY})
    {\archtitle Medverse: context-fused, dual U-Net cross-attn.};

  % ---- inner step boxes: Fuse, Dual U-Net ----
  \draw[black!30,fill=white,rounded corners=5pt,line width=0.8pt]
    (\xFuse,\yCtx-\clusterhalf-\boxpad) rectangle ({\xFuse+\boxw},\yTgt+\clusterhalf+\boxpad);
  \draw[black!30,fill=white,rounded corners=5pt,line width=0.8pt]
    ({\xEnc},\yCtx-\clusterhalf-\boxpad) rectangle ({\xEnc+\boxw},\yTgt+\clusterhalf+\boxpad);

  \node[anchor=south] at ({\boxFuseCenter},\yTgt+\clusterhalf+\boxpad+0.20) {\archboxlabel Fuse (context only)};
  \node[anchor=south] at ({\boxEncCenter},\yTgt+\clusterhalf+\boxpad+0.20) {\archboxlabel Dual U-Net (separate weights)};

  \archarrow{({\xFuse+\boxw+0.50},{\yMid})}{({\xEnc-0.50},{\yMid})}{black!30}

  % ---- column headers (Fuse box) ----
  \node[anchor=south,font=\normalsize] at (\ximgTop,\yTgt+\clusterhalf+0.25) {img};
  \node[anchor=south,font=\normalsize] at (\xlblTop,\yTgt+\clusterhalf+0.25) {lbl};
  \node[anchor=south,font=\normalsize] at (\xFusedTop,\yTgt+\clusterhalf+0.25) {fused};

  % ---- background column groupings (Fuse box) ----
  \foreach \x in {\ximgTop, \xlblTop, \xFusedTop} {
    \fill[black!15,rounded corners=3pt]
      ({\x-\tsz/2-0.18},{\yCtx-\toff-\tsz/2-0.10}) rectangle
      ({\x+\tsz/2+0.18},{\yTgt+\toff+\tsz/2+0.10});
  }

  % ---- target row: raw image tokens, unsplit fill, bypass the fuse ----
  \tpcluster{\ximgTop}{\yTgt}{archTargetGreen}{black!70}
  \tpcluster{\xFusedTop}{\yTgt}{archTargetGreen}{black!70}
  \fusearrow{({\ximgTop+\tsz/2+0.18},{\yTgt})}{({\xFusedTop-\tsz/2-0.18},{\yTgt})}{black!40}

  % ---- context row: img + lbl -> (+) -> fused ----
  \tpcluster{\ximgTop}{\yCtx}{archContextPink}{black!70}
  \tpcluster{\xlblTop}{\yCtx}{archContextAccent}{black!70}
  \tpclustersplit{\xFusedTop}{\yCtx}{archContextPink}{archContextAccent}{black!70}
  \node[font=\huge] at ({\xFuseOp},{\yCtx}) {$\oplus$};
  \fusearrow{({\xFuseArrowStart},{\yCtx})}{({\xFusedTop-\tsz/2-0.18},{\yCtx})}{black!55}

  % ---- Dual U-Net box: one bowtie per row (own weights, own resolution
  % schedule), coupled only at the waist -- narrowing half on the left,
  % widening half on the right, matching how \archbowtie is built ----
  \archarrow{({\xEnc-0.15},{\yTgt})}{({\bcx-0.05},{\yTgt})}{black!30}
  \archarrow{({\xEnc-0.15},{\yCtx})}{({\bcx-0.05},{\yCtx})}{black!30}
  \archbowtie{\bcx}{\bcyTgt}{\bw}{\bh}{\bpinch}{archTargetGreen}{target U-Net}
  \archbowtie{\bcx}{\bcyCtx}{\bw}{\bh}{\bpinch}{archContextPink}{context U-Net}

  % the pinch is at each row's own center height (yTgt/yCtx) by construction
  % (bcy = row center - bh/2), so the waist itself is the natural anchor
  \waistarrow{({\waistX},{\yTgt})}{({\waistX},{\yCtx})}{archBrightBlue}

  \node[align=center] at ({\boxEncCenter},{\outerBot+0.15})
    {\archnote context queries target only while narrowing; target queries context only\\
     while widening -- drawn here as one coupling for legibility};

  % ---- predicted mask output (target row only -- context's decoder output
  % is internal, never exposed) ----
  \archarrow{({\bcx+\bw+0.15},{\yTgt})}{({\xOut-\tsz/2-0.25},{\yTgt})}{archTargetAccent}
  \tpcluster{\xOut}{\yTgt}{archTargetGreen!40!black}{black!70}
  \node[anchor=north,font=\normalsize] at (\xOut,\yTgt-\clusterhalf-0.20) {pred.\ mask};

\end{tikzpicture}
