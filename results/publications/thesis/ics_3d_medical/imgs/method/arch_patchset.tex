% Dataflow arrows in this figure all use archArrowPale (a pale-green tint
% from arch-macros.tex); override it to light grey here, local to this
% standalone compilation only -- arch_medverse.tex also uses archArrowPale
% but compiles separately, so its own color is untouched.
\colorlet{archArrowPale}{black!30}

% Bigger arrows than the shared default (local override only -- other
% arch_*.tex figures keep the thinner \archarrow from arch-macros.tex).
\renewcommand{\archarrow}[3]{%
  \draw[-{Latex[length=3.6mm,width=3.2mm]},line width=3.2pt,#3] #1 -- #2;
}

% Same dual-attention arrow vocabulary as arch_fusion_axis.tex's "Dual
% attention" panel: \dblarrow for the Modality-axis pairwise edges,
% \tabpfnattn for the Position-axis fully-connected fan.
\definecolor{archBrightBlue}{HTML}{3B82F6}

% \dblarrow{p1}{p2}{color} -- curved double-headed attention edge.
% Thinner arrowheads than arch_fusion_axis.tex's original -- Position
% axis's fan (below) packs many overlapping arrows into a tight space,
% and the bigger heads made individual edges impossible to tell apart.
\newcommand{\dblarrow}[3]{
  \draw[{Latex[length=1.8mm,width=1.5mm]}-{Latex[length=1.8mm,width=1.5mm]},
        line width=1.5pt,#3] #1 to[bend left=12] #2;
}

% \tabpfnattn{x} -- fully connected, right-side routed attention with
% cascading transparency. Expects \tsz/\toff/\yTgt/\yCtx defined to match
% whichever token grid it is anchored to.
\newcommand{\tabpfnattn}[1]{
  \def\attnstyle{[{Latex[length=1.8mm,width=1.5mm]}-{Latex[length=1.8mm,width=1.5mm]}, line width=1.5pt, archBrightBlue]}
  % #1 is the lbl column's \archtokengrid x -- its LEFT edge, not a center
  % (unlike arch_fusion_axis.tex's tpsq/tpcluster, which are center-
  % anchored) -- so the fan must start past x + the drawn square's own
  % width (0.82*\tsz), not past x + half of it.
  \def\ax{#1+0.82*\tsz+0.05}

  \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=30] (\ax, \yTgt);
  \draw\attnstyle (\ax, \yTgt)       to[bend left=30] (\ax, \yTgt-\toff);
  \draw\attnstyle (\ax, \yTgt-\toff) to[bend left=30] (\ax, \yCtx+\toff);
  \draw\attnstyle (\ax, \yCtx+\toff) to[bend left=30] (\ax, \yCtx);
  \draw\attnstyle (\ax, \yCtx)       to[bend left=30] (\ax, \yCtx-\toff);

  \begin{scope}[opacity=0.6]
    \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=45] (\ax, \yTgt-\toff);
    \draw\attnstyle (\ax, \yTgt)       to[bend left=45] (\ax, \yCtx+\toff);
    \draw\attnstyle (\ax, \yTgt-\toff) to[bend left=45] (\ax, \yCtx);
    \draw\attnstyle (\ax, \yCtx+\toff) to[bend left=45] (\ax, \yCtx-\toff);
  \end{scope}

  \begin{scope}[opacity=0.3]
    \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=60] (\ax, \yCtx+\toff);
    \draw\attnstyle (\ax, \yTgt)       to[bend left=60] (\ax, \yCtx);
    \draw\attnstyle (\ax, \yTgt-\toff) to[bend left=60] (\ax, \yCtx-\toff);
  \end{scope}

  \begin{scope}[opacity=0.15]
    \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=75] (\ax, \yCtx);
    \draw\attnstyle (\ax, \yTgt)       to[bend left=75] (\ax, \yCtx-\toff);
  \end{scope}

  \begin{scope}[opacity=0.08]
    \draw\attnstyle (\ax, \yTgt+\toff) to[bend left=90] (\ax, \yCtx-\toff);
  \end{scope}
}
